cmake_minimum_required(VERSION 3.24)
project(Win10NativePatch VERSION 2.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
add_library(PatchRuntime STATIC PatchRuntime.cpp Touchpad.cpp)
target_include_directories(PatchRuntime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(PatchRuntime PUBLIC user32 shell32 ole32 oleaut32 advapi32 bcrypt comctl32 shlwapi)
add_executable(NativeSnap WIN32 NativeSnap.cpp NativeSnap.rc)
target_link_libraries(NativeSnap PRIVATE PatchRuntime dwmapi gdi32)
add_executable(Win10PatchOptions WIN32 Options.cpp Options.rc)
target_link_libraries(Win10PatchOptions PRIVATE PatchRuntime uxtheme dwmapi)
add_library(ExplorerTabs SHARED ExplorerTabs.cpp ExplorerTabs.def)
target_link_libraries(ExplorerTabs PRIVATE PatchRuntime dwmapi uxtheme gdi32 uuid propsys oleacc uiautomationcore)
add_executable(NativePatchSetup WIN32 Setup.cpp Setup.rc LegacyMigration.cpp)
target_link_libraries(NativePatchSetup PRIVATE PatchRuntime msi version xmllite)
add_executable(NativePatchModelTests ModelTests.cpp)
foreach(target NativeSnap Win10PatchOptions NativePatchSetup ExplorerTabs)
  if(MSVC)
    target_link_options(${target} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/NativePatch.manifest")
  endif()
endforeach()
