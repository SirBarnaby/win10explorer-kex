cmake_minimum_required(VERSION 3.24)
project(Win10NativePatch VERSION 2.0.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
add_library(PatchRuntime STATIC PatchRuntime.cpp Touchpad.cpp)
target_include_directories(PatchRuntime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(PatchRuntime PUBLIC user32 shell32 ole32 oleaut32 advapi32 bcrypt comctl32 shlwapi)
add_executable(NativeSnap WIN32 NativeSnap.cpp NativeSnap.rc)
target_link_libraries(NativeSnap PRIVATE PatchRuntime dwmapi gdi32)
add_executable(Win10PatchOptions WIN32 Options.cpp Options.rc)
target_link_libraries(Win10PatchOptions PRIVATE PatchRuntime uxtheme dwmapi)
add_library(HeaderLayout STATIC HeaderLayout.cpp TabMotion.cpp CaptionInk.cpp)
target_link_libraries(HeaderLayout PUBLIC user32 dwmapi advapi32)
add_executable(HeaderLayoutTests HeaderLayoutTests.cpp)
target_link_libraries(HeaderLayoutTests PRIVATE HeaderLayout)
add_executable(MotionTests MotionTests.cpp)
target_link_libraries(MotionTests PRIVATE HeaderLayout)
add_executable(CaptionInkTests CaptionInkTests.cpp)
target_link_libraries(CaptionInkTests PRIVATE HeaderLayout)
add_executable(TabModelTests TabModelTests.cpp)
add_executable(HeaderHookOwnershipTests HeaderHookOwnershipTests.cpp)
target_link_libraries(HeaderHookOwnershipTests PRIVATE HeaderPublicApiHooks)
add_library(HeaderAllocationHook SHARED EXCLUDE_FROM_ALL HeaderAllocationTests.cpp HeaderIntegration.cpp)
target_compile_definitions(HeaderAllocationHook PRIVATE HEADER_ALLOCATION_DLL)
target_link_libraries(HeaderAllocationHook PRIVATE HeaderLayout HeaderPublicApiHooks user32 advapi32 comctl32 uxtheme dwmapi bcrypt)
add_executable(HeaderAllocationTests EXCLUDE_FROM_ALL HeaderAllocationTests.cpp)
target_link_libraries(HeaderAllocationTests PRIVATE user32 advapi32)
add_library(ExplorerTabs SHARED ExplorerTabs.cpp HeaderIntegration.cpp TabHeaderRenderer.cpp ExplorerTabs.def)
target_link_libraries(ExplorerTabs PRIVATE PatchRuntime HeaderLayout HeaderPublicApiHooks dwmapi uxtheme gdi32 uuid propsys oleacc uiautomationcore dcomp d3d11 dxgi d2d1 dwrite windowscodecs)
add_executable(HeaderRendererStateTests EXCLUDE_FROM_ALL HeaderRendererStateTests.cpp TabHeaderRenderer.cpp)
target_link_libraries(HeaderRendererStateTests PRIVATE HeaderLayout user32 ole32 shell32 gdiplus gdi32 uxtheme dcomp d3d11 dxgi d2d1 dwrite windowscodecs)
add_executable(ExplorerMvpHost WIN32 ExplorerMvpHost.cpp)
target_link_libraries(ExplorerMvpHost PRIVATE PatchRuntime dwmapi)
# Sandbox-only copy of the starter: the Windows Sandbox guest is build 19041.
add_executable(ExplorerMvpHostLab WIN32 EXCLUDE_FROM_ALL ExplorerMvpHost.cpp)
target_compile_definitions(ExplorerMvpHostLab PRIVATE NATIVE_LAB_ANY_BUILD=1)
target_link_libraries(ExplorerMvpHostLab PRIVATE PatchRuntime dwmapi)
add_executable(ExplorerHeaderProbe WIN32 ExplorerHeaderProbe.cpp)
target_link_libraries(ExplorerHeaderProbe PRIVATE user32 ole32 oleacc oleaut32 shell32)
add_executable(NativePatchSetup WIN32 Setup.cpp Setup.rc LegacyMigration.cpp)
target_link_libraries(NativePatchSetup PRIVATE PatchRuntime msi version xmllite)
add_executable(ExplorerMvpSetup WIN32 MvpSetup.cpp MvpSetup.rc LegacyMigration.cpp)
target_link_libraries(ExplorerMvpSetup PRIVATE PatchRuntime uuid xmllite)
add_executable(NativePatchModelTests ModelTests.cpp)
add_library(SnapLayout STATIC SnapLayout.cpp)
target_link_libraries(SnapLayout PUBLIC user32 advapi32)
target_link_libraries(NativeSnap PRIVATE SnapLayout)
add_executable(SnapLayoutTests SnapLayoutTests.cpp)
target_link_libraries(SnapLayoutTests PRIVATE SnapLayout)
add_executable(SnapIntegrationTests SnapIntegrationTests.cpp)
target_link_libraries(SnapIntegrationTests PRIVATE SnapLayout user32 advapi32 dwmapi)
add_library(HeaderLabHook SHARED HeaderLab.cpp)
target_compile_definitions(HeaderLabHook PRIVATE HEADER_LAB_DLL)
target_link_libraries(HeaderLabHook PRIVATE user32 advapi32 comctl32 ole32 oleaut32 uuid)
add_executable(HeaderLab HeaderLab.cpp)
target_link_libraries(HeaderLab PRIVATE user32 advapi32 comctl32 gdiplus oleacc ole32 oleaut32 uxtheme dwmapi)
add_library(HeaderInteractionHook SHARED EXCLUDE_FROM_ALL HeaderInteractionTests.cpp)
target_compile_definitions(HeaderInteractionHook PRIVATE HEADER_INTERACTION_DLL)
target_link_libraries(HeaderInteractionHook PRIVATE user32 advapi32)
add_executable(HeaderInteractionTests EXCLUDE_FROM_ALL HeaderInteractionTests.cpp)
target_link_libraries(HeaderInteractionTests PRIVATE user32 advapi32 oleacc ole32 oleaut32 uuid gdiplus gdi32 dwmapi)
add_executable(HeaderBehaviorTests EXCLUDE_FROM_ALL HeaderBehaviorTests.cpp)
target_link_libraries(HeaderBehaviorTests PRIVATE user32 advapi32 oleacc ole32 oleaut32 shell32 uuid gdiplus gdi32 dwmapi)
add_library(HeaderMaterialHook SHARED EXCLUDE_FROM_ALL HeaderMaterialProbe.cpp)
target_compile_definitions(HeaderMaterialHook PRIVATE HEADER_MATERIAL_DLL)
add_library(HeaderPublicApiHooks STATIC
  vendor/minhook/src/buffer.c
  vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c
  vendor/minhook/src/hde/hde64.c)
target_include_directories(HeaderMaterialHook PRIVATE vendor/minhook/include)
target_link_libraries(HeaderMaterialHook PRIVATE HeaderPublicApiHooks)
target_link_libraries(HeaderMaterialHook PRIVATE user32 advapi32 comctl32 uxtheme dcomp d3d11 dxgi d2d1 dwrite)
add_executable(HeaderMaterialProbe EXCLUDE_FROM_ALL HeaderMaterialProbe.cpp)
target_link_libraries(HeaderMaterialProbe PRIVATE user32 advapi32 gdiplus)
if(HEADER_PROOF_SOURCE)
  add_library(ExplorerTabsSurfaceProof SHARED "${HEADER_PROOF_SOURCE}" HeaderIntegration.cpp ExplorerTabs.def)
  target_link_libraries(ExplorerTabsSurfaceProof PRIVATE PatchRuntime HeaderLayout HeaderPublicApiHooks dwmapi uxtheme gdi32 uuid propsys oleacc uiautomationcore dcomp d3d11 dxgi d2d1 dwrite)
  target_include_directories(ExplorerTabsSurfaceProof PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  set_target_properties(ExplorerTabsSurfaceProof PROPERTIES OUTPUT_NAME ExplorerTabs RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/candidate")
endif()
foreach(target NativeSnap Win10PatchOptions NativePatchSetup ExplorerTabs ExplorerMvpHost ExplorerMvpSetup)
  if(MSVC)
    target_link_options(${target} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/NativePatch.manifest")
  endif()
endforeach()
